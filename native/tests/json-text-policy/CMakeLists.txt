cmake_minimum_required(VERSION 3.21)
project(PandoeditorJsonTextPolicyCheck LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Core)
include(CTest)
get_filename_component(REPOSITORY_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_executable(json_text_policy_tests ../json_text_policy_tests.cpp
    "${REPOSITORY_ROOT}/shared/core/src/losslessjson.cpp")
target_include_directories(json_text_policy_tests PRIVATE "${REPOSITORY_ROOT}/shared/core/include")
target_link_libraries(json_text_policy_tests PRIVATE Qt6::Core)
if(MSVC)
    target_compile_options(json_text_policy_tests PRIVATE /UNDEBUG /utf-8)
else()
    target_compile_options(json_text_policy_tests PRIVATE -UNDEBUG)
endif()
add_test(NAME json_text_policy_tests COMMAND json_text_policy_tests)
set_tests_properties(json_text_policy_tests PROPERTIES TIMEOUT 10)
