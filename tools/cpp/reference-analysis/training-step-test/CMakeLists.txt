cmake_minimum_required(VERSION 3.20)
project(pando_reference_training_step LANGUAGES CXX)
find_package(Torch REQUIRED)
separate_arguments(PANDO_TORCH_FLAGS NATIVE_COMMAND "${TORCH_CXX_FLAGS}")
add_library(reference-training-test-core STATIC ../reference-model.cpp ../boundary-loss.cpp ../training-checkpoint.cpp)
target_compile_features(reference-training-test-core PUBLIC cxx_std_20)
target_compile_options(reference-training-test-core PUBLIC ${PANDO_TORCH_FLAGS})
target_link_libraries(reference-training-test-core PUBLIC ${TORCH_LIBRARIES})
add_executable(reference-training-step-test ../training-step-tests.cpp)
target_link_libraries(reference-training-step-test PRIVATE reference-training-test-core)
add_executable(reference-training-ordered-test ../training-ordered-tests.cpp)
target_link_libraries(reference-training-ordered-test PRIVATE reference-training-test-core)
add_executable(reference-training-amp-test ../training-amp-tests.cpp)
target_link_libraries(reference-training-amp-test PRIVATE reference-training-test-core)
add_executable(pando-reference-training-batch ../training-batch.cpp ../reference-io.cpp ../safetensors.cpp ../../../../shared/core/src/losslessjson.cpp)
target_include_directories(pando-reference-training-batch PRIVATE ../../../../shared/core/include)
target_link_libraries(pando-reference-training-batch PRIVATE reference-training-test-core)
add_executable(reference-training-infer-fixture ../training-infer-fixture.cpp ../reference-io.cpp)
target_link_libraries(reference-training-infer-fixture PRIVATE reference-training-test-core)
