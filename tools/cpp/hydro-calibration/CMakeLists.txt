cmake_minimum_required(VERSION 3.20)
project(pando_hydro_calibration LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(pando-hydro-calibration ../hydro/calibration.cpp ../../../shared/core/src/losslessjson.cpp)
target_include_directories(pando-hydro-calibration PRIVATE ../../../shared/core/include)
if(MSVC)
    target_compile_options(pando-hydro-calibration PRIVATE /utf-8)
else()
    target_compile_options(pando-hydro-calibration PRIVATE -Wall -Wextra -Werror -Wno-error=misleading-indentation -pedantic)
endif()
