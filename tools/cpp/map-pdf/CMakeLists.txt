cmake_minimum_required(VERSION 3.21)
project(pando_map_pdf LANGUAGES C)
if(NOT PANDO_MUPDF_SDK_ROOT)
    message(FATAL_ERROR "PANDO_MUPDF_SDK_ROOT must name the pinned MuPDF SDK")
endif()
add_executable(pando-map-pdf pdf-read.c)
target_include_directories(pando-map-pdf PRIVATE "${PANDO_MUPDF_SDK_ROOT}/mupdf-devel/include")
if(WIN32)
    target_link_libraries(pando-map-pdf PRIVATE "${PANDO_MUPDF_SDK_ROOT}/mupdf-devel/lib/mupdfcpp64.lib")
else()
    target_link_libraries(pando-map-pdf PRIVATE "${PANDO_MUPDF_SDK_ROOT}/libmupdf.so.26.11" m)
    set_target_properties(pando-map-pdf PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN")
endif()
